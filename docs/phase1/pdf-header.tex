\newcommand{\prahariTableFont}{\fontsize{9.3}{11.1}\selectfont}
% Body tables never split across pages: every row end acts as \\* (no page break) and the whole
% table is one chunk. \AppendixTables lets the long appendix index break between rows again.
\makeatletter
\let\prahariLTcrOrig\LT@tabularcr
\protected\def\prahariLTcrKeep{%
  \relax\iffalse{\fi\ifnum0=`}\fi
  \def\crcr{\LT@crcr\noalign{\nobreak}}\let\cr\crcr
  \LT@t@bularcr}
\let\LT@tabularcr\prahariLTcrKeep
\newcommand{\AppendixTables}{\renewcommand{\prahariTableFont}{\fontsize{9.8}{11.8}\selectfont}%
  \let\LT@tabularcr\prahariLTcrOrig}
% \KeepTables turns the no-split rule back on (for the short reproducibility card)
\newcommand{\KeepTables}{\let\LT@tabularcr\prahariLTcrKeep}
% longtable's \LT@start resets \@colroom (the text height left after floats) to the full column
% height, so LaTeX then places a float where there is no room and it runs off the page. A kept
% table never breaks, so keep LaTeX's own \@colroom for it.
\def\prahariMaybeColroom{\ifx\LT@tabularcr\prahariLTcrKeep\else\global\@colroom\@colht\fi}
\ExplSyntaxOn
\patchcmd{\LT@start}{\global\@colroom\@colht}{\prahariMaybeColroom}{}
  {\PackageWarning{prahari}{longtable~colroom~patch~failed}}
\ExplSyntaxOff
\makeatother
\setcounter{LTchunksize}{100}
% A table that continues on a new page repeats its caption as "Table N. (continued)" above the
% same header row. Pandoc puts the caption only in the first head; the rows after \endfirsthead
% form the continuation head, so the caption is added at its start. \caption[] steps no counter
% and adds no list entry. Short body tables never break, so they never show it.
\makeatletter
\def\endfirsthead{\LT@end@hd@ft\LT@firsthead\caption[]{(continued)}\tabularnewline}
\makeatother
% The last foot (bottom rule) is also used as the page foot, so its height is reserved on every page.
% Without this, a table that ends exactly at the foot of a page left an empty repeated header on
% the next page.
% (\LT@end@hd@ft saves the box and then calls \LT@get@widths once; hook that call.)
\makeatletter
\let\prahariLTgw\LT@get@widths
% The hook is set inside \noalign: a bare \def here would open an empty (shaded) table row.
\def\endlastfoot{%
  \noalign{\gdef\LT@get@widths{\setbox\LT@foot\copy\LT@lastfoot
    \global\let\LT@get@widths\prahariLTgw\LT@get@widths}}%
  \LT@end@hd@ft\LT@lastfoot}
\makeatother
% Row shading extends \tabcolsep beyond the outer cells; narrowing the table by 2\tabcolsep (it is
% centred) keeps the shading and rules inside the text block.
% Inside a longtable the rules are plain \hrule lines in \noalign, which span the full text width,
% so they line up exactly with the row shading.
\newcommand{\prahariLTrule}[4]{\noalign{\vskip#3{\color{#2}\hrule height #1}\vskip#4}}

% \refurl{host/path} prints https://host/path as a link. "https://" is boxed, so the URL never
% breaks inside or after the scheme; inside the rest it breaks only after / . _ ? = & (never at a
% hyphen or inside a word). A space before it gives the line a normal break point.
\newcommand{\refurl}[1]{\begingroup\def\UrlBreaks{\do\/\do\.\do\_\do\?\do\=\do\&}\def\UrlBigBreaks{}%
  \def\UrlOrds{\do\*\do\-\do\~\do\'\do\"\do\:}%
  \href{https://#1}{\mbox{https://}\nolinkurl{#1}}\endgroup}
% File paths break only after "/" (never inside a word); hashes may break anywhere.
\DeclareUrlCommand\path{\urlstyle{tt}\def\UrlBreaks{\do\/}\def\UrlBigBreaks{}\def\UrlNoBreaks{}}
\DeclareUrlCommand\hash{\urlstyle{tt}}
